#这只是一个示例
#真实输入时不支持导入txt
#以下是格式

size = somethings #空格可有可不有，这个代表了格式，由normal（普通，即纯色）或pro（多色）
#normal 版本
size.color(x)#x代表参数，可以自己写RGB，三个参数，当然，由默认参数，如：red,blue,green,yellow（共10个默认参数）
#pro 版本
size.x_y(x,y)#代表了像素值，此时假设为3,3
size.pro.x = [    #x是一个自定义的名字，随机,
[[red],[green],[blue]],
[[255,255,255],[100,203,102],[111,111,0]],
[[#你应该可以根据上面的两种推断出来了],[],[]]
]
use(size.pro.x)#该行代表使用这个list
